Comandos disponibles:

// Ver el sistema
sistema

// Crear usuarios
sistema.agregarUsuario("pepe@pepe.es")
sistema.agregarUsuario("luis@luis.es")
sistema.agregarUsuario("ana@ana.es")

// Ver usuarios
sistema.usuarios

// Crear partida
codigo1 = sistema.crearPartida("pepe@pepe.es")

// Ver partidas
sistema.partidas

// Ver partidas disponibles
sistema.obtenerPartidasDisponibles()

// Unir a partida
sistema.unirAPartida("luis@luis.es", codigo1)

// Ver jugadores de la partida
sistema.partidas[codigo1].jugadores

// Intentar unir un tercero (debería fallar - partida llena)
sistema.unirAPartida("ana@ana.es", codigo1)

// Crear otra partida
codigo2 = sistema.crearPartida("ana@ana.es")

// Ver todas las partidas disponibles
sistema.obtenerPartidasDisponibles()
